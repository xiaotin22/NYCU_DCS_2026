module GE(
    input clk,
    input rst_n,
    input in_valid,
    input [15:0] in_data_eq0,
    input [15:0] in_data_eq1,
    input [15:0] in_data_eq2,
    output logic out_valid,
    output logic signed [5:0] out_data0,
    output logic signed [5:0] out_data1,
    output logic signed [5:0] out_data2,
    output logic [1:0] exception
);

logic st;


logic [1:0] ex_r;
logic signed [2:0] a0_r, b0_r, c0_r;
logic signed [6:0] d0_r;
logic signed [5:0] x2_r, x3_r;

wire signed [2:0] a0 = in_data_eq0[2:0];
wire signed [2:0] b0 = in_data_eq0[5:3];
wire signed [2:0] c0 = in_data_eq0[8:6];
wire signed [6:0] d0 = in_data_eq0[15:9];
wire signed [2:0] a1 = in_data_eq1[2:0];
wire signed [2:0] b1 = in_data_eq1[5:3];
wire signed [2:0] c1 = in_data_eq1[8:6];
wire signed [6:0] d1 = in_data_eq1[15:9];
wire signed [2:0] a2 = in_data_eq2[2:0];
wire signed [2:0] b2 = in_data_eq2[5:3];
wire signed [2:0] c2 = in_data_eq2[8:6];
wire signed [6:0] d2 = in_data_eq2[15:9];

function automatic signed [5:0] mul3_3(input signed [2:0] k, input signed [2:0] v);
    begin
        mul3_3 = k * v;
    end
endfunction

function automatic signed [9:0] mul3_7(input signed [2:0] k, input signed [6:0] v);
    begin
        mul3_7 = k * v;
    end
endfunction

function automatic signed [9:0] mul3_6(input signed [2:0] k, input signed [5:0] v);
    begin
        mul3_6 = k * v;
    end
endfunction



function automatic [3:0] tz11(input signed [10:0] v);
    begin
        if (v[0])       tz11 = 4'd0;
        else if (v[1])  tz11 = 4'd1;
        else if (v[2])  tz11 = 4'd2;
        else if (v[3])  tz11 = 4'd3;
        else if (v[4])  tz11 = 4'd4;
        else if (v[5])  tz11 = 4'd5;
        else if (v[6])  tz11 = 4'd6;
        else if (v[7])  tz11 = 4'd7;


        else if (v[8])  tz11 = 4'd8;
        else if (v[9])  tz11 = 4'd9;
        else            tz11 = 4'd10;
    end
endfunction

function automatic [5:0] ashr6_10(input signed [9:0] v, input [2:0] sh);
    begin
        case (sh)
            3'd0:  ashr6_10 = v[5:0];
            3'd1:  ashr6_10 = v[6:1];
            3'd2:  ashr6_10 = v[7:2];


            3'd3:  ashr6_10 = v[8:3];
            3'd4:  ashr6_10 = v[9:4];
            3'd5:  ashr6_10 = {v[9], v[9:5]};
            3'd6:  ashr6_10 = {{2{v[9]}}, v[9:6]};


            default: ashr6_10 = {{3{v[9]}}, v[9:7]};
        endcase
    end
endfunction

function automatic [5:0] mul_mod6(input [5:0] a, input [5:0] b);
    logic p00, p01, p02, p03, p04, p05;
    logic p10, p11, p12, p13, p14;
    logic p20, p21, p22, p23;
    logic p30, p31, p32;
    logic p40, p41, p50;
    logic s2a, c1a, c2a, c2b;
    logic s3a, s3b, c3a, c3b, c3c;
    logic s4a, s4b, s4c, c4a, c4b, c4c, c4d;
    begin
        p00 = a[0] & b[0];
        p01 = a[0] & b[1];
        p02 = a[0] & b[2];
        p03 = a[0] & b[3];
        p04 = a[0] & b[4];
        p05 = a[0] & b[5];
        p10 = a[1] & b[0];
        p11 = a[1] & b[1];
        p12 = a[1] & b[2];
        p13 = a[1] & b[3];
        p14 = a[1] & b[4];
        p20 = a[2] & b[0];
        p21 = a[2] & b[1];
        p22 = a[2] & b[2];
        p23 = a[2] & b[3];
        p30 = a[3] & b[0];
        p31 = a[3] & b[1];
        p32 = a[3] & b[2];
        p40 = a[4] & b[0];
        p41 = a[4] & b[1];
        p50 = a[5] & b[0];

        mul_mod6[0] = p00;
        mul_mod6[1] = p01 ^ p10;
        c1a = p01 & p10;

        s2a = p02 ^ p11 ^ p20;
        c2a = (p02 & p11) | (p02 & p20) | (p11 & p20);
        mul_mod6[2] = s2a ^ c1a;
        c2b = s2a & c1a;

        s3a = p03 ^ p12 ^ p21;
        c3a = (p03 & p12) | (p03 & p21) | (p12 & p21);
        s3b = p30 ^ c2a ^ c2b;
        c3b = (p30 & c2a) | (p30 & c2b) | (c2a & c2b);
        mul_mod6[3] = s3a ^ s3b;
        c3c = s3a & s3b;

        s4a = p04 ^ p13 ^ p22;
        c4a = (p04 & p13) | (p04 & p22) | (p13 & p22);
        s4b = p31 ^ p40 ^ c3a;
        c4b = (p31 & p40) | (p31 & c3a) | (p40 & c3a);
        s4c = c3b ^ c3c ^ s4a;
        c4c = (c3b & c3c) | (c3b & s4a) | (c3c & s4a);
        mul_mod6[4] = s4b ^ s4c;
        c4d = s4b & s4c;

        mul_mod6[5] = p05 ^ p14 ^ p23 ^ p32 ^ p41 ^ p50 ^ c4a ^ c4b ^ c4c ^ c4d;
    end
endfunction



function automatic [5:0] inv_odd6(input [5:0] v);
    begin
        case (v)
            6'd1:  inv_odd6 = 6'd1;
            6'd3:  inv_odd6 = 6'd43;
            6'd5:  inv_odd6 = 6'd13;
            6'd7:  inv_odd6 = 6'd55;
            6'd9:  inv_odd6 = 6'd57;
            6'd11: inv_odd6 = 6'd35;
            6'd13: inv_odd6 = 6'd5;
            6'd15: inv_odd6 = 6'd47;
            6'd17: inv_odd6 = 6'd49;
            6'd19: inv_odd6 = 6'd27;
            6'd21: inv_odd6 = 6'd61;
            6'd23: inv_odd6 = 6'd39;
            6'd25: inv_odd6 = 6'd41;
            6'd27: inv_odd6 = 6'd19;
            6'd29: inv_odd6 = 6'd53;
            6'd31: inv_odd6 = 6'd31;
            6'd33: inv_odd6 = 6'd33;
            6'd35: inv_odd6 = 6'd11;
            6'd37: inv_odd6 = 6'd45;
            6'd39: inv_odd6 = 6'd23;
            6'd41: inv_odd6 = 6'd25;
            6'd43: inv_odd6 = 6'd3;
            6'd45: inv_odd6 = 6'd37;
            6'd47: inv_odd6 = 6'd15;
            6'd49: inv_odd6 = 6'd17;
            6'd51: inv_odd6 = 6'd59;
            6'd53: inv_odd6 = 6'd29;
            6'd55: inv_odd6 = 6'd7;


            6'd57: inv_odd6 = 6'd9;
            6'd59: inv_odd6 = 6'd51;
            6'd61: inv_odd6 = 6'd21;
            default: inv_odd6 = 6'd63;
        endcase
    end
endfunction

function automatic signed [5:0] div_a0_fast(input signed [9:0] num, input signed [2:0] den);
    begin
        case (den)
            3'b001: div_a0_fast = num[5:0];
            3'b111: div_a0_fast = -num[5:0];
            3'b010: div_a0_fast = ashr6_10(num, 3'd1);


            3'b110: div_a0_fast = -ashr6_10(num, 3'd1);
            3'b011: div_a0_fast = mul_mod6(num[5:0], 6'd43);
            3'b101: div_a0_fast = mul_mod6(num[5:0], 6'd21);
            default: div_a0_fast = -ashr6_10(num, 3'd2);
        endcase
    end
endfunction

wire signed [5:0] r1b = mul3_3(a0, b1) - mul3_3(a1, b0);
wire signed [5:0] r1c = mul3_3(a0, c1) - mul3_3(a1, c0);
wire signed [9:0] r1d = mul3_7(a0, d1) - mul3_7(a1, d0);
wire signed [5:0] r2b = mul3_3(a0, b2) - mul3_3(a2, b0);
wire signed [5:0] r2c = mul3_3(a0, c2) - mul3_3(a2, c0);
wire signed [9:0] r2d = mul3_7(a0, d2) - mul3_7(a2, d0);

wire signed [10:0] det_p0 = r1b * r2c;
wire signed [10:0] det_p1 = r2b * r1c;
wire signed [10:0] det2 = det_p0 - det_p1;

wire signed [15:0] num2 = (r1d * r2c) - (r2d * r1c);
wire signed [15:0] num3 = (r2d * r1b) - (r1d * r2b);

wire r1_bad = !(|r1b) && !(|r1c) && (|r1d);
wire r2_bad = !(|r2b) && !(|r2c) && (|r2d);
wire infinite = !(|num2) && !(|num3) && !r1_bad && !r2_bad;
wire [1:0] ex_w = !(|det2) ? (infinite ? 2'b01 : 2'b10) : 2'b00;

wire [3:0] det_sh = tz11(det2);
logic [5:0] det_div, n2_div, n3_div;

always_comb begin
    case (det_sh)
        4'd0: begin
            det_div = det2[5:0];
            n2_div = num2[5:0];
            n3_div = num3[5:0];
        end
        4'd1: begin
            det_div = det2[6:1];
            n2_div = num2[6:1];
            n3_div = num3[6:1];
        end
        4'd2: begin
            det_div = det2[7:2];
            n2_div = num2[7:2];
            n3_div = num3[7:2];
        end
        4'd3: begin
            det_div = det2[8:3];
            n2_div = num2[8:3];
            n3_div = num3[8:3];
        end
        4'd4: begin
            det_div = det2[9:4];
            n2_div = num2[9:4];
            n3_div = num3[9:4];
        end
        4'd5: begin
            det_div = det2[10:5];
            n2_div = num2[10:5];
            n3_div = num3[10:5];
        end
        4'd6: begin
            det_div = {det2[10], det2[10:6]};
            n2_div = num2[11:6];
            n3_div = num3[11:6];
        end
        4'd7: begin
            det_div = {{2{det2[10]}}, det2[10:7]};
            n2_div = num2[12:7];
            n3_div = num3[12:7];
        end
        4'd8: begin
            det_div = {{3{det2[10]}}, det2[10:8]};
            n2_div = num2[13:8];
            n3_div = num3[13:8];
        end
        4'd9: begin
            det_div = {{4{det2[10]}}, det2[10:9]};
            n2_div = num2[14:9];
            n3_div = num3[14:9];
        end
        default: begin
            det_div = 6'd1;
            n2_div = 6'd0;
            n3_div = 6'd0;
        end
    endcase
end

wire [5:0] det_inv = inv_odd6(det_div);
wire signed [5:0] x2_w = mul_mod6(n2_div, det_inv);
wire signed [5:0] x3_w = mul_mod6(n3_div, det_inv);
wire signed [9:0] x1_num = {{3{d0_r[6]}}, d0_r} - mul3_6(b0_r, x2_r) - mul3_6(c0_r, x3_r);
wire signed [5:0] x1_w = div_a0_fast(x1_num, a0_r);


wire out_ok = out_valid && !(|ex_r);

assign exception = out_valid ? ex_r : 2'b00;
assign out_data0 = out_ok ? x1_w : 6'sd0;
assign out_data1 = out_ok ? x2_r : 6'sd0;
assign out_data2 = out_ok ? x3_r : 6'sd0;

always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        st <= 1'b0;
        out_valid <= 1'b0;
    end else begin
        out_valid <= 1'b0;
        if (st) begin
            st <= 1'b0;
            out_valid <= 1'b1;
        end else if (in_valid) begin


            st <= 1'b1;
            a0_r <= a0;
            b0_r <= b0;
            c0_r <= c0;
            d0_r <= d0;
            x2_r <= x2_w;
            x3_r <= x3_w;
            ex_r <= ex_w;
        end
    end
end

endmodule
